# Every workspace member's runtime level, one entry per line: the crate, its level, and the cargo
# feature flags its normal dependency tree is read with; a crate checked at two feature sets has
# two entries. `scripts/runtime_levels.py` fails CI on an entry off its level and on a member with
# no entry, so a new crate is classified before it merges. The levels are the transports DESIGN's
# §8, "Two levels of runtime independence":
#
#   runtime-free           no `tokio` crate in the tree, so a plug point is implemented without it.
#   runtime-neutral        tokio only with `default` (empty) and `sync`, which hyper brings; no
#                          reactor, executor or timer, so the crate runs on any runtime.
#   runtime-neutral+http2  the same, plus `io-util` and `bytes`, which h2 brings on the HTTP/2 path.
#   tokio-bound            runs on tokio; listed so no member goes unclassified, and not checked.
#
# In both runtime-neutral sets tokio arrives only through hyper, h2 or tokio-util: a package that
# depends on tokio directly fails the entry unless it is one of those three, and a workspace member
# that depends on tokio-util directly fails it unless a line `<crate> direct-tokio-util <reason>`
# names the member. A `direct-tokio-util` line no runtime-neutral tree uses fails too.
#
# A runtime-free crate is read with `--all-features` unless a feature of its own brings tokio, so a
# feature added later is checked without this file changing.

# The core: timing through `Timer`, spawning through `Spawn`, every DI and lifecycle plug point.
ulo                           runtime-free           --all-features

# The core's attribute macros, compiled into every application's tree.
ulo-macros                    runtime-free           --all-features

# The handler protocol every transport's attribute macro is written against.
ulo-handler-codegen           runtime-free           --all-features

# The transport-neutral call, reply and error traits.
ulo-transport                 runtime-free           --all-features

# The derives for `ulo-transport`'s error and validation traits.
ulo-transport-macros          runtime-free           --all-features

# `std::net` sockets, socket activation and TLS up to rustls's `ServerConfig`.
ulo-net                       runtime-free           --all-features

# The second runtime, on `async-executor` and `async-io`.
ulo-smol                      runtime-free           --all-features

# The runtime suite, which any `Runtime` implementation runs.
ulo-runtime-conformance       runtime-free           --all-features

# Certificates generated with `rcgen` for the TLS tests, reached through dev-dependencies alone.
ulo-test-certs                runtime-free           --all-features

# The HTTP hub, which spawns nothing, at default features: its `tokio-io` feature brings tokio's
# I/O traits for `Upgraded`, and with them `tokio` with `sync`.
ulo-http                      runtime-free

# The HTTP route attributes.
ulo-http-macros               runtime-free           --all-features

# The WebSocket hub: `async-tungstenite` over `futures-io`, every task on the app's `Runtime`.
ulo-ws                        runtime-free           --all-features

# The WebSocket gateway attributes.
ulo-ws-macros                 runtime-free           --all-features

# The WebSocket suite, its client over the host's `futures-io` stream on the host's runtime.
ulo-ws-conformance            runtime-free           --all-features

# The RPC hub, where the `Link` SPI lives, on the app's `Runtime`.
ulo-rpc                       runtime-free           --all-features

# The RPC pattern attributes.
ulo-rpc-macros                runtime-free           --all-features

# The TCP wire over `futures-io`, on the runtime a link hands it.
ulo-rpc-tcp-core              runtime-free           --all-features

# The smol TCP link: `async-net`'s sockets and `futures-rustls` over `ulo-rpc-tcp-core`.
ulo-rpc-tcp-smol              runtime-free           --all-features

# The RPC suite at default features: its `tokio` feature brings the tokio harness.
ulo-rpc-conformance           runtime-free

# The RPC suite with its smol harness, as the smol CI job runs it.
ulo-rpc-conformance           runtime-free           --features smol

# The transport-neutral GraphQL hub, where the `Engine` SPI lives.
ulo-graphql                   runtime-free           --all-features

# GraphQL over HTTP, which depends on `ulo-graphql-ws` and so on `ulo-ws`.
ulo-graphql-http              runtime-free           --all-features

# graphql-transport-ws, spawning through `Connection::runtime`.
ulo-graphql-ws                runtime-free           --all-features

# The async-graphql engine; async-graphql at the features it is used with brings no runtime.
ulo-graphql-async-graphql     runtime-free           --all-features

# The juniper engine; juniper brings no runtime.
ulo-graphql-juniper           runtime-free           --all-features

# The gRPC method attribute, a proc-macro crate with no runtime of its own.
ulo-grpc-macros               runtime-free           --all-features

# The gRPC build-script code generator, run at build time with no runtime of its own.
ulo-build                     runtime-free           --all-features

# The accept loop and the `Listener` plug point, on hyper with no features: tokio with `sync`.
ulo-hyper-serve               runtime-neutral        --all-features

# The smol listener, adopting `ulo-net`'s sockets behind `ulo-hyper-serve`.
ulo-listen-smol               runtime-neutral        --all-features

# The WebSocket server without its default `tokio` feature, which brings `ulo-listen-tokio`;
# HTTP/1.1 alone, so it reads as `ulo-hyper-serve` does.
ulo-ws-hyper                  runtime-neutral        --no-default-features

# The same crate at its default features, on the tokio listener.
ulo-ws-hyper                  tokio-bound

# The HTTP backend without its default `tokio` feature, which brings `ulo-listen-tokio`; it serves
# HTTP/2, so h2 brings `io-util` and `bytes`.
ulo-http-hyper                runtime-neutral+http2  --no-default-features

# The same crate at its default features, on the tokio listener.
ulo-http-hyper                tokio-bound

# The HTTP suite at default features, its client hyper's and h2's: its `tokio` and `rocket`
# features bring their hosts' runtimes.
ulo-http-conformance          runtime-neutral+http2

# The HTTP suite with its smol harness, as the smol CI job runs it.
ulo-http-conformance          runtime-neutral+http2  --features smol

# The suite's GOAWAY scenario drives `h2`'s client over the host's `futures-io` stream, and h2 reads
# tokio's I/O traits; the adapter between them is tokio-util's or one implementing tokio's traits,
# which would name tokio directly.
ulo-http-conformance          direct-tokio-util      `compat`, the host's stream as the tokio I/O h2's client reads
